## 1. 单位系统与转换

### 1.1 核心常量

```typescript
// 96 DPI 标准：1 inch = 96 pixels, 1 inch = 25.4 mm
// PX_PER_MM = 96 / 25.4 = 3.779527559
const PX_PER_MM = 3.779527559
const MM_TO_PX = 3.779527559
```

### 1.2 转换函数

```typescript
/**
 * mm → px 转换
 * @param mm 毫米值
 * @returns 像素值
 */
function mmToPx(mm: number): number {
  return mm * MM_TO_PX
}

/**
 * px → mm 转换
 * @param px 像素值
 * @returns 毫米值
 */
function pxToMm(px: number): number {
  return px / PX_PER_MM
}

/**
 * DPI 转换（用于公式渲染等高清晰度需求）
 * @param mm 毫米值
 * @param dpi 目标 DPI
 * @returns 像素值
 */
function mmToPxByDpi(mm: number, dpi: number = 96): number {
  return mm * (dpi / 25.4)
}
```

### 1.3 字体宽度估算

```typescript
/**
 * 估算单行文本宽度
 * @param text 文本内容
 * @param fontSize 字号（px）
 * @param isChinese 是否包含中文
 * @returns 估算宽度（px）
 */
function estimateTextWidth(text: string, fontSize: number, isChinese: boolean): number {
  if (isChinese) {
    return text.length * fontSize
  }
  return text.length * fontSize * 0.6
}
```

## 5. 自动排版算法

### 5.1 水平紧凑排版 (applyCompactLayout)

```typescript
/**
 * 水平紧凑排版算法
 * @param cards 卡片列表（会被直接修改）
 * @param pageWidth 页面宽度（px）
 * @param margins 页边距（mm）
 * @param cardMinHSpacing 最小水平间距（mm）
 * @param cardMinVSpacing 最小垂直间距（mm）
 */
function applyCompactLayout(
  cards: Card[],
  pageWidth: number,
  margins: { top: number; bottom: number; left: number; right: number },
  cardMinHSpacing: number,
  cardMinVSpacing: number
): void {
  const pageContentWidth = mmToPx(pageWidth - margins.left - margins.right)
  const hSpacing = mmToPx(cardMinHSpacing)
  const vSpacing = mmToPx(cardMinVSpacing)

  // 过滤需要参与排版的卡片
  const layoutCards = cards
    .filter(c => !c.flags.excludeFromLayout)
    .sort((a, b) => a.order - b.order)

  if (layoutCards.length === 0) return

  let currentX = mmToPx(margins.left)
  let currentY = mmToPx(margins.top)
  let rowMaxH = 0
  const hSp = hSpacing

  for (const card of layoutCards) {
    const cardW = mmToPx(card.size.projectWidth)
    const cardH = mmToPx(card.size.projectHeight)

    // 换行判断
    if (currentX + cardW > pageContentWidth + mmToPx(margins.right)) {
      currentX = mmToPx(margins.left)
      currentY += rowMaxH + vSpacing
      rowMaxH = 0
    }

    card.position.x = currentX
    card.position.y = currentY
    currentX += cardW + hSp
    rowMaxH = Math.max(rowMaxH, cardH)
  }
}
```

### 5.2 竖直紧凑排版 (applyCompactLayoutVertical)

```typescript
/**
 * 竖直紧凑排版算法
 * @param cards 卡片列表（会被直接修改）
 * @param pageHeight 页面高度（px）
 * @param margins 页边距（mm）
 * @param cardMinHSpacing 最小水平间距（mm）
 * @param cardMinVSpacing 最小垂直间距（mm）
 */
function applyCompactLayoutVertical(
  cards: Card[],
  pageHeight: number,
  margins: { top: number; bottom: number; left: number; right: number },
  cardMinHSpacing: number,
  cardMinVSpacing: number
): void {
  const pageContentHeight = mmToPx(pageHeight - margins.top - margins.bottom)
  const hSpacing = mmToPx(cardMinHSpacing)
  const vSpacing = mmToPx(cardMinVSpacing)

  const layoutCards = cards
    .filter(c => !c.flags.excludeFromLayout)
    .sort((a, b) => a.order - b.order)

  if (layoutCards.length === 0) return

  let currentX = mmToPx(margins.left)
  let currentY = mmToPx(margins.top)
  let colMaxH = 0
  let colCount = 0

  for (const card of layoutCards) {
    const cardW = mmToPx(card.size.projectWidth)
    const cardH = mmToPx(card.size.projectHeight)

    // 新列判断
    if (currentY + cardH > pageContentHeight + mmToPx(margins.bottom)) {
      colCount++
      currentX += colCount * cardW + hSpacing * colCount
      currentY = mmToPx(margins.top)
    }

    card.position.x = currentX
    card.position.y = currentY
    currentY += cardH + vSpacing
  }
}
```

### 5.3 碰撞检测与吸附

```typescript
const SNAP_THRESHOLD = 5 // 吸附阈值（px）

interface SnapResult {
  x: number
  y: number
  snappedX: boolean
  snappedY: boolean
}

function snapToEdges(
  x: number,
  y: number,
  card: Card,
  otherCards: Card[],
  pageWidth: number,
  pageHeight: number,
  margins: { top: number; bottom: number; left: number; right: number }
): SnapResult {
  let resultX = x
  let resultY = y
  let snappedX = false
  let snappedY = false

  const cardW = mmToPx(card.size.projectWidth)
  const cardH = mmToPx(card.size.projectHeight)
  const pageW = mmToPx(pageWidth)
  const pageH = mmToPx(pageHeight)

  // 安全区域边缘
  const safeEdges = {
    left: mmToPx(margins.left),
    right: pageW - mmToPx(margins.right) - cardW,
    centerX: pageW / 2 - cardW / 2,
    top: mmToPx(margins.top),
    bottom: pageH - mmToPx(margins.bottom) - cardH,
    centerY: pageH / 2 - cardH / 2,
  }

  // X 方向吸附
  for (const edge of [safeEdges.left, safeEdges.right, safeEdges.centerX]) {
    if (Math.abs(x - edge) < SNAP_THRESHOLD) {
      resultX = edge
      snappedX = true
      break
    }
  }

  // Y 方向吸附
  for (const edge of [safeEdges.top, safeEdges.bottom, safeEdges.centerY]) {
    if (Math.abs(y - edge) < SNAP_THRESHOLD) {
      resultY = edge
      snappedY = true
      break
    }
  }

  // 卡片边缘吸附
  for (const other of otherCards) {
    if (other.id === card.id) continue
    const otherW = mmToPx(other.size.projectWidth)
    const otherH = mmToPx(other.size.projectHeight)

    // 右边缘对齐
    if (Math.abs(x - (other.position.x + otherW)) < SNAP_THRESHOLD) {
      resultX = other.position.x + otherW
      snappedX = true
    }
    // 左边缘对齐
    if (Math.abs(x + cardW - other.position.x) < SNAP_THRESHOLD) {
      resultX = other.position.x - cardW
      snappedX = true
    }
    // 上边缘对齐
    if (Math.abs(y - (other.position.y + otherH)) < SNAP_THRESHOLD) {
      resultY = other.position.y + otherH
      snappedY = true
    }
    // 下边缘对齐
    if (Math.abs(y + cardH - other.position.y) < SNAP_THRESHOLD) {
      resultY = other.position.y - cardH
      snappedY = true
    }
  }

  return { x: resultX, y: resultY, snappedX, snappedY }
}
```

```typescript
function getNumberPrefix(index: number, style?: TitleNumberStyle): string {
  switch (style) {
    case 'number':
      return `${index}`
    case 'number-dot':
      return `${index}.`
    case 'number-hash':
      return `${index}# `
    case 'number-hash-prefix':
      return `#${index}`
    case 'number-dash':

    case 'number-space':
      return `${index} `
    default:
```

## 7. 富文本与公式

### 7.1 公式渲染

```typescript
import katex from 'katex'
import 'katex/dist/katex.min.css'

/**
 * 渲染 LaTeX 公式为 HTML 字符串
 * @param formula LaTeX 公式
 * @param displayMode 是否为独立公式（display mode）
 * @returns HTML 字符串
 */
function renderLatex(formula: string, displayMode: boolean = false): string {
  try {
    return katex.renderToString(formula, {
      displayMode,
      throwOnError: false,
      strict: false,
    })
  } catch (error) {
    console.error('LaTeX render error:', error)
    return `<span class="latex-error">${formula}</span>`
  }
}

/**
 * 从 TipTap JSON 提取所有公式
 * @param content TipTap JSON
 * @returns 公式列表
 */
function extractFormulas(content: object): Array<{ formula: string; display: boolean }> {
  const formulas: Array<{ formula: string; display: boolean }> = []
  const text = JSON.stringify(content)

  // 匹配 $$...$$（独立公式）
  const displayRegex = /\$\$([^\n$]+?)\$\$/g
  let match
  while ((match = displayRegex.exec(text)) !== null) {
    formulas.push({ formula: match[1], display: true })
  }

  // 匹配 $...$（行内公式）
  const inlineRegex = /\$([^\n$]+?)\$/g
  while ((match = inlineRegex.exec(text)) !== null) {
    formulas.push({ formula: match[1], display: false })
  }

  return formulas
}
```

### 7.2 富文本解析

```typescript
interface TextNode {
  type: 'text'
  text: string
  marks?: Array<{ type: string; attrs?: Record<string, unknown> }>
}

interface ParagraphNode {
  type: 'paragraph'
  content?: TextNode[]
}

interface HeadingNode {
  type: 'heading'
  attrs?: { level: number }
  content?: TextNode[]
}

/**
 * 解析 TipTap JSON 为可显示的文本行
 * @param content TipTap JSON
 * @returns 文本行数组
 */
function parseRichText(content: object): string[] {
  const lines: string[] = []

  function traverse(node: unknown): void {
    if (!node || typeof node !== 'object') return

    const n = node as Record<string, unknown>

    if (n.type === 'paragraph' || n.type === 'heading') {
      const paragraphText = extractText(n.content)
      if (paragraphText.trim()) {
        lines.push(paragraphText)
      }
    } else if (n.type === 'doc' && Array.isArray(n.content)) {
      n.content.forEach(traverse)
    }
  }

  traverse(content)
  return lines
}

function extractText(content: unknown): string {
  if (!content || typeof content !== 'object') return ''
  if (!Array.isArray(content)) return ''

  return content.map((item) => {
    if (typeof item === 'object' && item !== null && 'text' in item) {
      return (item as TextNode).text || ''
    }
    return ''
  }).join('')
}
```

### 9.1 核心功能实现

```typescript
// RegexReplaceDialog.tsx

interface RegexReplaceState {
  inputText: string
  pattern: string
  replace: string
  outputText: string
  matchCount: number
  matches: Array<{ start: number; end: number; text: string }>
}

/**
 * 执行正则替换并返回匹配信息
 */
function executeRegexReplace(
  inputText: string,
  pattern: string,
  replace: string
): { outputText: string; matchCount: number; matches: Array<{ start: number; end: number; text: string }> } {
  if (!pattern || !inputText) {
    return { outputText: inputText, matchCount: 0, matches: [] }
  }

  try {
    const regex = new RegExp(pattern, 'g')
    const matches: Array<{ start: number; end: number; text: string }> = []

    // 查找所有匹配
    let match
    while ((match = regex.exec(inputText)) !== null) {
      matches.push({
        start: match.index,
        end: match.index + match[0].length,
        text: match[0],
      })
      // 防止无限循环
      if (match[0].length === 0) {
        regex.lastIndex++
      }
    }

    // 执行替换
    const actualReplace = replace.replace(/\\n/g, '\n')
    const outputText = inputText.replace(regex, actualReplace)

    return {
      outputText,
      matchCount: matches.length,
      matches,
    }
  } catch (error) {
    console.error('Regex error:', error)
    return { outputText: inputText, matchCount: 0, matches: [] }
  }
}

/**
 * 合并连续空行
 */
function mergeBlankLines(text: string): string {
  return text.replace(/\n{3,}/g, '\n\n')
}
```

```typescript
// auth.ts

const SALT = 'knowcard-salt-2025'

/**
 * SHA-256 哈希（使用 Web Crypto API）
 */
async function hashPassword(password: string): Promise<string> {
  const encoder = new TextEncoder()
  const data = encoder.encode(SALT + password)
  const hashBuffer = await crypto.subtle.digest('SHA-256', data)
  const hashArray = Array.from(new Uint8Array(hashBuffer))
  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('')
}

/**
 * 验证密码
 */
async function verifyPassword(password: string, hash: string): Promise<boolean> {
  const passwordHash = await hashPassword(password)
  return passwordHash === hash
}
```

```typescript
    const next = new Set(prev)

      next.delete(id)
    } else {
      next.add(id)
    }

  })
```

```typescript
async function handleBatchExport(): Promise<void> {
  if (selectedProjectIds.size === 0) {
    alert('请先选择要导出的项目')
    return
  }

  const selectedProjects = projects.filter(p => selectedProjectIds.has(p.id))
  const allData = []

  // 遍历选中项目，获取每个项目的数据
  for (const project of selectedProjects) {
    const blob = await exportProject(project.id)
    const text = await blob.text()
    const data = JSON.parse(text)
    allData.push(data)
  }

  // 构建合并数据
  const mergedData = {
    version: 2,
    type: 'multi-project',
    projects: allData,
    exportedAt: new Date().toISOString()
  }

  const mergedBlob = new Blob([JSON.stringify(mergedData, null, 2)], { type: 'application/json' })
  const url = URL.createObjectURL(mergedBlob)
  const a = document.createElement('a')
  a.href = url

  // 生成文件名
  const dateStr = new Date().toISOString().slice(0, 10).replace(/-/g, '')
  const names = selectedProjects.map(p => p.name).join('_')

  // 文件名超长处理：超过25字符截断
  let displayNames = names
  if (names.length > 25) {
    displayNames = names.substring(0, 25) + `…等${selectedProjects.length}个文件`
  }

  a.download = `KnowCard_${displayNames}_${dateStr}.json`
  a.click()
  URL.revokeObjectURL(url)
}
```
